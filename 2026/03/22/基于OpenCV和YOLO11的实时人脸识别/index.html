该项目基于OpenCV内置的目标检测功能和YOLO的人脸识别功能
我将以机器学习的基本的三个环节，分板块执行：
get_picture.py——准备数据
train.py—————训练模型
start.py—————开始识别

获取图片
需要导入的库如下：

import cv2
import os
import tkinter as tk
from tkinter import simpledialog
import shutil

定义camera：打开摄像头并初始化，分辨率以一般笔记本的标准 1280*720。

def camera():
    cap = cv2.VideoCapture(0)
    cap.set(cv2.CAP_PROP_FRAME_WIDTH, 1280)
    cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 720)

    if not cap.isOpened():
        print("can't open the webcam")
        cap.release()
        exit()
    return cap

定义path：创建所需路径和文件夹，train为训练集图片的位置，val为验证集图片的位置

def path():
    root_dir = "dataset"
    face_train = os.path.join(root_dir, "train")
    face_val = os.path.join(root_dir, "val")
    os.makedirs(face_train, exist_ok=True)
    os.makedirs(face_val, exist_ok=True)
    return face_train,face_val

定义name：弹出窗口输入个人信息，即图片的标注（输入后才能继续，避免出现文件路径混乱）

def name(face_train):
    root = tk.Tk()
    root.withdraw()
    user_name = ""
    while True:
        user_name = simpledialog.askstring("命名", "请输入姓名")
        if user_name is None:
            exit()
        elif user_name.strip() == "":
            print("请输入姓名")
            continue
        else:
            user_name = user_name.strip()
            break

    user_face_dir = os.path.join(face_train, user_name)
    os.makedirs(user_face_dir, exist_ok=True)
    return user_face_dir,user_name

定义detect：按下s保存摄像头图片后，使用OpenCV内置目标检测器CascadeClassifier实现人脸目标检测，并裁剪出人脸的图片。根据裁剪位置信息，先复制摄像头原图片文件再裁剪，可以避免rectangle绘制的识别框被裁入，降低杂乱数据干扰。

def detect(cap,user_face_dir):
    face_counter = 0
    face_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + 'haarcascade_frontalface_default.xml')
    while True:
        ret, frame = cap.read()
        if not ret:
            break

        #人脸检测
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        faces = face_cascade.detectMultiScale(gray, 1.1, 4)

        origin_frame=frame.copy()     #防止绿框被截取

        #绘制人脸框
        for (x, y, w, h) in faces:
            cv2.rectangle(frame, (x, y), (x+w, y+h), (0, 255, 0), 2)

        cv2.imshow('Webcam', frame)
        k = cv2.waitKey(1)

        # ESC退出
        if k % 256 == 27:
            print("Escape hit closing")
            break

        elif k % 256 == ord('s'):      #按s保存人脸裁剪图（最好不少与20张）
            if len(faces) > 0:
                x, y, w, h = faces[0]
                face_crop = origin_frame[y:y+h, x:x+w]
                face_name = os.path.join(user_face_dir, f"face_{face_counter}.png")
                cv2.imwrite(face_name, face_crop)
                print(f"人脸保存: {face_name}")
                face_counter += 1

定义split：数据分类函数，将保存在train里面的图片文件，按比例转移到val文件夹。比例可根据数据量具体更改，此处用0.2。

def split(face_train,face_val, user_name):
    current_train = os.path.join(face_train, user_name)
    current_val = os.path.join(face_val, user_name)
    os.makedirs(current_val, exist_ok=True)

    val_images = [f for f in os.listdir(current_val) if f.endswith(('png','jpg','jpeg'))]

    if len(val_images) > 0:
        return

    img_list = [f for f in os.listdir(current_train) if f.endswith(('png','jpg','jpeg'))]
    total = len(img_list)

    if total == 0:
        return

    split_num = max(1, int(total * 0.2)) #比例为0.2
    for img in img_list[:split_num]:
        shutil.move(os.path.join(current_train, img), os.path.join(current_val, img))

最后使用release和destroyAllWindows函数释放内存，至此数据准备完毕。

cap=camera()
face_train,face_val=path()
user_face_dir,user_name = name(face_train)
detect(cap,user_face_dir)
split(face_train,face_val,user_name)
cap.release()
cv2.destroyAllWindows()

开始训练
使用适合人脸检测的yolo11n-cls模型训练，根据电脑处理器选择device，此处使用cpu。
训练完成后就会将结果保存至run/runs/classify/face_recognize/weights

from ultralytics import YOLO

model = YOLO("yolo11n-cls.pt")

model.train(
    data="./dataset",    
    epochs=30,
    imgsz=224,
    batch=4, 
    device="cpu",
    name="face_recognize",
    val=True
    )

开始识别
模型选择best.pt文件，打开摄像头开始人脸识别，识别出的分数会打印在窗口左上角。

from ultralytics import YOLO
import cv2
import os

model = YOLO("runs/classify/face_recognize/weights/best.pt")

cap = cv2.VideoCapture(0)

while True:
    ret, frame = cap.read()
    if not ret:
        break

    results = model(frame)
    cv2.imshow("人脸识别", results[0].plot())

    if cv2.waitKey(1) == 27:
        break

cap.release()
cv2.destroyAllWindows()

总结
本项目基于 OpenCV 与 YOLO 算法实现基础视觉应用，整体代码简洁轻量化。依托 YOLO 算法的高效特性，项目仅通过极简代码便实现了优异的目标识别效果。目前项目仍存在可优化方向：摄像头采集的图像清晰度不足、数据处理流程有待精细化，同时还存在异常图片数据需手动清理的缺陷。通过本次实战开发，我对 OpenCV 与 YOLO 两项技术的理论认知与工程实践能力均得到了显著提升。
